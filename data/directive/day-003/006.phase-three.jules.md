import { z } from 'zod';
import {
  GeneratorModel,
  STUDIO_DEFAULT_VIEWPORT_OPTIONS,
} from '../generator.model.js';
import type GeneratorBit from '../fce/generator.bit.js';
import type {
  ViewportCompilationOptions,
  ViewportOptionRegistryEntry,
} from '../fce/generator.interface.js';
import type State from '../../99.core/state.js';

export class ViewportOptionValidationError extends Error {
  constructor(message: string) {
    super(`:: [FAIL] [GENERATOR:CONFIG] ${message}`);
    this.name = 'ViewportOptionValidationError';
  }
}

export const ResolutionTupleSchema = z
  .object({
    width: z.number().int().positive().refine((w) => w === 1920 || w === 1280 || w === 3840, {
      message: 'Unsupported canvas width: must be 1280, 1920 (1080p studio default), or 3840',
    }),
    height: z.number().int().positive().refine((h) => h === 1080 || h === 720 || h === 2160, {
      message: 'Unsupported canvas height: must be 720, 1080 (1080p studio default), or 2160',
    }),
  })
  .strict()
  .refine(
    (res) =>
      (res.width === 1920 && res.height === 1080) ||
      (res.width === 1280 && res.height === 720) ||
      (res.width === 3840 && res.height === 2160),
    {
      message: 'Invalid resolution tuple aspect ratio: must be 16:9 (1920x1080, 1280x720, or 3840x2160)',
    }
  );

export const FrameSliceRangeSchema = z
  .object({
    startFrame: z.number().int().min(0),
    endFrame: z.number().int().min(0),
  })
  .strict()
  .refine((slice) => slice.endFrame >= slice.startFrame, {
    message: 'Frame slice endFrame must be greater than or equal to startFrame',
  });

export const ViewportCompilationOptionsSchema = z
  .object({
    episodeId: z
      .string()
      .min(2)
      .max(32)
      .regex(/^[a-z0-9_-]+$/i, 'episodeId must contain only alphanumeric, underscore, or hyphen characters'),
    sceneId: z
      .string()
      .min(2)
      .max(32)
      .regex(/^[a-z0-9_-]+$/i, 'sceneId must contain only alphanumeric, underscore, or hyphen characters'),
    seriesSlug: z
      .string()
      .min(3)
      .max(64)
      .regex(/^[a-z0-9]+(?:-[a-z0-9]+)*$/, 'seriesSlug must conform to POSIX kebab-case'),
    resolution: ResolutionTupleSchema,
    fps: z.literal(24, {
      errorMap: () => ({ message: 'Studio timebase invariant violation: fps must equal 24' }),
    }),
    glBackend: z.enum(['swangle', 'angle']),
    frameSlice: FrameSliceRangeSchema.optional(),
  })
  .strict();

/**
 * Merges partial user overrides with STUDIO_DEFAULT_VIEWPORT_OPTIONS and
 * enforces fail-closed schema validation before any downstream unit runs.
 */
export const validateViewportOptions = (
  rawInput?: Partial<ViewportCompilationOptions>
): ViewportCompilationOptions => {
  const candidate = {
    episodeId: rawInput?.episodeId ?? STUDIO_DEFAULT_VIEWPORT_OPTIONS.episodeId,
    sceneId: rawInput?.sceneId ?? STUDIO_DEFAULT_VIEWPORT_OPTIONS.sceneId,
    seriesSlug: rawInput?.seriesSlug ?? STUDIO_DEFAULT_VIEWPORT_OPTIONS.seriesSlug,
    resolution: rawInput?.resolution ?? STUDIO_DEFAULT_VIEWPORT_OPTIONS.resolution,
    fps: rawInput?.fps ?? STUDIO_DEFAULT_VIEWPORT_OPTIONS.fps,
    glBackend: rawInput?.glBackend ?? STUDIO_DEFAULT_VIEWPORT_OPTIONS.glBackend,
    ...(rawInput?.frameSlice !== undefined ? { frameSlice: rawInput.frameSlice } : {}),
  };

  const parsed = ViewportCompilationOptionsSchema.safeParse(candidate);
  if (!parsed.success) {
    const details = parsed.error.issues.map((i) => i.message).join('; ');
    throw new ViewportOptionValidationError(details);
  }

  return Object.freeze({
    ...parsed.data,
    resolution: Object.freeze({ ...parsed.data.resolution }),
    ...(parsed.data.frameSlice
      ? { frameSlice: Object.freeze({ ...parsed.data.frameSlice }) }
      : {}),
  });
};

export const initGenerator = (
  cpy: GeneratorModel,
  bal: GeneratorBit,
  _ste?: State
): GeneratorModel => {
  const defaultOptions = validateViewportOptions();
  const defaultEntry: ViewportOptionRegistryEntry = Object.freeze({
    optionKey: 'default-studio-1080p24',
    label: 'STUDIO DEFAULT (1920x1080 @ 24fps | swangle)',
    activeOptions: defaultOptions,
    registeredAtUtc: new Date().toISOString(),
  });

  cpy.activeOptions = defaultOptions;
  cpy.registeredPresets = {
    [defaultEntry.optionKey]: defaultEntry,
  };
  cpy.validationErrorCount = 0;
  cpy.lastValidatedAtUtc = defaultEntry.registeredAtUtc;

  if (bal.slv) {
    bal.slv({
      genBit: {
        idx: 'init-generator',
        val: 1,
        resolved: defaultOptions,
        entry: defaultEntry,
      },
    });
  }
  return cpy;
};

export const updateGenerator = (
  cpy: GeneratorModel,
  bal: GeneratorBit,
  _ste?: State
): GeneratorModel => {
  if (bal.idx) cpy.generatorBits[bal.idx] = bal;
  if (bal.slv) bal.slv({ genBit: bal });
  return cpy;
};

export const registerViewportOptions = (
  cpy: GeneratorModel,
  bal: GeneratorBit,
  _ste?: State
): GeneratorModel => {
  let validated: ViewportCompilationOptions;
  try {
    validated = validateViewportOptions(bal.dat);
  } catch (err) {
    cpy.validationErrorCount += 1;
    throw err;
  }

  const optionKey =
    bal.idx && bal.idx.trim().length > 0
      ? bal.idx.trim()
      : `${validated.seriesSlug}:${validated.episodeId}:${validated.sceneId}`;

  const label =
    bal.src && bal.src.trim().length > 0
      ? bal.src.trim()
      : `${validated.episodeId}/${validated.sceneId} (${validated.resolution.width}x${validated.resolution.height} @ ${validated.fps}fps | ${validated.glBackend})`;

  const entry: ViewportOptionRegistryEntry = Object.freeze({
    optionKey,
    label,
    activeOptions: validated,
    registeredAtUtc: new Date().toISOString(),
  });

  cpy.registeredPresets[optionKey] = entry;
  cpy.activeOptions = validated;
  cpy.lastValidatedAtUtc = entry.registeredAtUtc;

  if (bal.slv) {
    bal.slv({
      genBit: {
        idx: optionKey,
        src: label,
        val: 1,
        resolved: validated,
        entry,
      },
    });
  }

  return cpy;
};

export const selectViewportPreset = (
  cpy: GeneratorModel,
  bal: GeneratorBit,
  _ste?: State
): GeneratorModel => {
  const existing = bal.idx ? cpy.registeredPresets[bal.idx] : undefined;
  if (!existing) {
    cpy.validationErrorCount += 1;
    throw new ViewportOptionValidationError(
      `Unknown viewport option preset key: '${bal.idx ?? ''}'`
    );
  }

  cpy.activeOptions = existing.activeOptions;
  cpy.lastValidatedAtUtc = new Date().toISOString();

  if (bal.slv) {
    bal.slv({
      genBit: {
        idx: existing.optionKey,
        src: existing.label,
        val: 1,
        resolved: existing.activeOptions,
        entry: existing,
      },
    });
  }

  return cpy;
};

import { describe, it, expect, vi } from 'vitest';
import {
  GeneratorModel,
  STUDIO_DEFAULT_VIEWPORT_OPTIONS,
} from '../01.generator.unit/generator.model.js';
import {
  validateViewportOptions,
  registerViewportOptions,
  selectViewportPreset,
  initGenerator,
  ViewportOptionValidationError,
} from '../01.generator.unit/buz/viewport.config.js';

describe('01.generator.unit :: Viewport Option Selector Unit Registration', () => {
  it('resolves studio default fallbacks (1920x1080, 24fps, swangle) when no overrides are given', () => {
    const resolved = validateViewportOptions();
    expect(resolved).toEqual(STUDIO_DEFAULT_VIEWPORT_OPTIONS);
    expect(resolved.resolution).toEqual({ width: 1920, height: 1080 });
    expect(resolved.fps).toBe(24);
    expect(resolved.glBackend).toBe('swangle');
  });

  it('throws ViewportOptionValidationError immediately on invalid resolution tuples', () => {
    expect(() =>
      validateViewportOptions({
        resolution: { width: 1920, height: 720 } as any,
      })
    ).toThrow(ViewportOptionValidationError);

    expect(() =>
      validateViewportOptions({
        resolution: { width: -100, height: 1080 } as any,
      })
    ).toThrow(ViewportOptionValidationError);
  });

  it('throws ViewportOptionValidationError when fps violates the 24fps studio invariant', () => {
    expect(() =>
      validateViewportOptions({
        fps: 60 as any,
      })
    ).toThrow(ViewportOptionValidationError);
  });

  it('registers named presets into GeneratorModel and switches active options cleanly', () => {
    const model = initGenerator(new GeneratorModel(), { idx: 'init' });
    const slv = vi.fn();

    registerViewportOptions(model, {
      idx: 'ep01-tracer-slice',
      dat: {
        episodeId: 'ep01',
        sceneId: 'sc02',
        seriesSlug: 'under-the-floorboards',
        frameSlice: { startFrame: 0, endFrame: 35 },
      },
      slv,
    });

    expect(model.registeredPresets['ep01-tracer-slice']).toBeDefined();
    expect(model.activeOptions.sceneId).toBe('sc02');
    expect(model.activeOptions.frameSlice).toEqual({ startFrame: 0, endFrame: 35 });
    expect(slv).toHaveBeenCalledWith(
      expect.objectContaining({
        genBit: expect.objectContaining({ idx: 'ep01-tracer-slice', val: 1 }),
      })
    );

    selectViewportPreset(model, { idx: 'default-studio-1080p24' });
    expect(model.activeOptions.sceneId).toBe('sc01');
  });
});
