### 5. Pure 7-Bit ASCII Sanitizer & Telemetry Stream Pipe Buzzer: `packages/100.agent/99.bus.unit/buz/bus.viewport.ts`

```typescript
import { BusModel } from '../bus.model.js';
import type BusBit from '../fce/bus.bit.js';
import type {
  TelemetryPacket,
  TelemetryEnvelope,
  CompilationStage,
} from '../fce/bus.interface.js';
import type State from '../../99.core/state.js';

const UPDATE_CONSOLE = '[Console action] Update Console';
const VALID_STAGES: ReadonlySet<CompilationStage> = new Set([
  'INIT',
  'CRAWL',
  'BAKE',
  'VENDOR',
  'SEAL',
]);

/**
 * Strips all characters outside printable 7-bit ASCII (0x20-0x7E)
 * and strips ANSI escape sequences to preserve curses buffer integrity.
 */
export const sanitizeTo7BitAscii = (
  raw: string
): { cleanAscii: string; strippedCount: number } => {
  if (!raw) {
    return { cleanAscii: '', strippedCount: 0 };
  }
  // Strip ANSI escape sequences first
  const withoutAnsi = raw.replace(/\x1B\[[0-9;]*[A-Za-z]/g, '');
  let strippedCount = raw.length - withoutAnsi.length;
  let cleanAscii = '';

  for (const char of withoutAnsi) {
    const code = char.codePointAt(0) ?? 0;
    if (code >= 0x20 && code <= 0x7e) {
      cleanAscii += char;
    } else {
      strippedCount += 1;
    }
  }

  return { cleanAscii, strippedCount };
};

/**
 * Formats a TelemetryPacket into a pure 7-bit ASCII status line.
 */
export const formatTelemetryLine = (
  pkt: TelemetryPacket
): { formattedLine: string; strippedCount: number } => {
  const stage: CompilationStage = VALID_STAGES.has(pkt.stage)
    ? pkt.stage
    : 'INIT';

  const prefix =
    pkt.level === 'FAIL'
      ? ':: [FAIL]'
      : pkt.level === 'WARN'
        ? ':: [WARN]'
        : '>> [OK]';

  const rawLine = `${prefix} [${stage}:${pkt.subsystem.toUpperCase()}] ${pkt.message}`;
  const { cleanAscii, strippedCount } = sanitizeTo7BitAscii(rawLine);

  return {
    formattedLine: cleanAscii,
    strippedCount,
  };
};

export const initBus = (
  cpy: BusModel,
  bal: BusBit,
  _ste?: State
): BusModel => {
  cpy.messagesEmittedCount = 0;
  cpy.nonAsciiBytesStrippedTotal = 0;
  cpy.ringBuffer = [];
  cpy.busBits = {};
  if (bal.slv) bal.slv({ busBit: { idx: 'init-bus', val: 1 } });
  return cpy;
};

export const updateBus = (
  cpy: BusModel,
  bal: BusBit,
  _ste?: State
): BusModel => {
  if (bal.idx) cpy.busBits[bal.idx] = bal;
  if (bal.slv) bal.slv({ busBit: bal });
  return cpy;
};

export const flushBusRing = (
  cpy: BusModel,
  bal: BusBit,
  _ste?: State
): BusModel => {
  cpy.ringBuffer = [];
  if (bal.slv) bal.slv({ busBit: { idx: 'flush-bus-ring', val: 1 } });
  return cpy;
};

/**
 * Routes sanitized 7-bit ASCII telemetry lines into cns00 via LIBRARY.hunt,
 * or falls back cleanly to stdout in headless environments.
 */
export const routeTelemetry = async (
  cpy: BusModel,
  bal: BusBit,
  _ste?: State
): Promise<BusModel> => {
  const pkt = bal.dat;
  if (!pkt) {
    if (bal.slv) bal.slv({ busBit: { idx: bal.idx, val: 0 } });
    return cpy;
  }

  const { formattedLine, strippedCount } = formatTelemetryLine(pkt);
  const sinkIdx = cpy.activeSinkIdx || 'cns00';

  const envelope: TelemetryEnvelope = {
    seq: cpy.messagesEmittedCount + 1,
    timestampUtc: pkt.timestampUtc ?? new Date().toISOString(),
    sinkIdx,
    stage: pkt.stage,
    level: pkt.level,
    formattedLine,
    sanitizedByteCount: Buffer.byteLength(formattedLine, 'utf8'),
    strippedNonAsciiCount: strippedCount,
  };

  cpy.messagesEmittedCount += 1;
  cpy.nonAsciiBytesStrippedTotal += strippedCount;

  if (cpy.ringBuffer.length >= cpy.maxRingBufferSize) {
    cpy.ringBuffer.shift();
  }
  cpy.ringBuffer.push(envelope);

  const lib = (globalThis as any).LIBRARY;
  if (lib && typeof lib.hunt === 'function') {
    await lib.hunt(UPDATE_CONSOLE, {
      idx: sinkIdx,
      src: formattedLine,
    });
  } else if (process.env.NODE_ENV !== 'test') {
    process.stdout.write(`${formattedLine}\n`);
  }

  if (bal.slv) {
    bal.slv({
      busBit: {
        idx: bal.idx,
        src: formattedLine,
        val: 1,
        envelope,
      },
    });
  }

  return cpy;
};

```
