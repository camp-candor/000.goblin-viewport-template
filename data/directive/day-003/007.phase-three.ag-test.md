### LEVEL 3: Negative-Control Validation, Studio Defaults & Preset Registry Probes

Execute live `tsx` probes asserting studio fallback resolution (`1920x1080`, `24fps`, `swangle`), fail-closed `ViewportOptionValidationError` rejection on malformed resolution/FPS/slug inputs, preset registration/switching, and immutable `clone-deep` reducer state isolation:

```bash
# 3.1 Studio Defaults & Negative-Control Fail-Closed Schema Guard Probe
npx tsx -e "
import {
  validateViewportOptions,
  ViewportOptionValidationError,
} from './packages/500.viewport/01.generator.unit/buz/viewport.config.js';
import { STUDIO_DEFAULT_VIEWPORT_OPTIONS } from './packages/500.viewport/01.generator.unit/generator.model.js';

// 1. Default studio invariants
const defaults = validateViewportOptions();
if (
  defaults.resolution.width !== 1920 ||
  defaults.resolution.height !== 1080 ||
  defaults.fps !== 24 ||
  defaults.glBackend !== 'swangle'
) {
  console.error(':: [FAIL] Level 3.1 Studio defaults deviated from 1920x1080 @ 24fps swangle:', defaults);
  process.exit(1);
}

// 2. Negative control: mismatched aspect ratio (1920x720)
let caughtRes = false;
try {
  validateViewportOptions({ resolution: { width: 1920, height: 720 } as any });
} catch (err) {
  if (err instanceof ViewportOptionValidationError) caughtRes = true;
}
if (!caughtRes) {
  console.error(':: [FAIL] Level 3.1 Failed to reject invalid resolution tuple 1920x720');
  process.exit(1);
}

// 3. Negative control: non-24fps timebase (60fps)
let caughtFps = false;
try {
  validateViewportOptions({ fps: 60 as any });
} catch (err) {
  if (err instanceof ViewportOptionValidationError) caughtFps = true;
}
if (!caughtFps) {
  console.error(':: [FAIL] Level 3.1 Failed to reject non-24fps timebase');
  process.exit(1);
}

// 4. Negative control: non-kebab-case seriesSlug
let caughtSlug = false;
try {
  validateViewportOptions({ seriesSlug: 'Invalid Series Slug!' });
} catch (err) {
  if (err instanceof ViewportOptionValidationError) caughtSlug = true;
}
if (!caughtSlug) {
  console.error(':: [FAIL] Level 3.1 Failed to reject invalid seriesSlug');
  process.exit(1);
}

console.log('>> [PASS] Level 3.1 Studio defaults and fail-closed negative controls verified.');
" || exit 1

# 3.2 Preset Registry, State Switching & Clone-Deep Reducer Isolation Probe
npx tsx -e "
import { GeneratorModel } from './packages/500.viewport/01.generator.unit/generator.model.js';
import { reducer } from './packages/500.viewport/01.generator.unit/generator.reduce.js';
import {
  InitGenerator,
  RegisterViewportOptions,
  SelectViewportPreset,
} from './packages/500.viewport/01.generator.unit/generator.action.js';

const baseModel = new GeneratorModel();
const initState = reducer(baseModel, new InitGenerator({ idx: 'boot' }));

if (!initState.registeredPresets['default-studio-1080p24']) {
  console.error(':: [FAIL] Level 3.2 Default studio preset not registered on init');
  process.exit(1);
}

const registeredState = reducer(
  initState,
  new RegisterViewportOptions({
    idx: 'ep01-tracer-bullet',
    dat: {
      episodeId: 'ep01',
      sceneId: 'sc03',
      seriesSlug: 'under-the-floorboards',
      frameSlice: { startFrame: 0, endFrame: 35 },
    },
  })
);

if (
  registeredState.activeOptions.sceneId !== 'sc03' ||
  registeredState.activeOptions.frameSlice?.endFrame !== 35 ||
  initState.activeOptions.sceneId !== 'sc01'
) {
  console.error(':: [FAIL] Level 3.2 Preset registration or clone-deep state isolation failed');
  process.exit(1);
}

const switchedBack = reducer(
  registeredState,
  new SelectViewportPreset({ idx: 'default-studio-1080p24' })
);

if (switchedBack.activeOptions.sceneId !== 'sc01') {
  console.error(':: [FAIL] Level 3.2 SelectViewportPreset failed to restore default-studio-1080p24');
  process.exit(1);
}

console.log('>> [PASS] Level 3.2 Preset registry, switching, and clone-deep reducer isolation verified.');
" || exit 1

echo ">> [LEVEL 3 PASSED] Option validation, negative controls, and preset registry certified."
```
