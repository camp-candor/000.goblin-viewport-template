export const STUDIO_FPS = 24 as const;
export const FRAME_DURATION_MS = 1000 / STUDIO_FPS; // 41.666666666666664 ms

export class TemporalQuantizationError extends Error {
  constructor(message: string) {
    super(`:: [FAIL] [TEMPORAL:QUANTIZE] ${message}`);
    this.name = 'TemporalQuantizationError';
  }

/**
 * Converts continuous millisecond duration into discrete 24 fps integer frames
 * using integer ceiling math and a mandatory 1-frame sub-frame clamp.
 */
export const quantizeDurationMsToFrames = (durationMs: number): number => {
  if (!Number.isFinite(durationMs) || durationMs < 0) {
    throw new TemporalQuantizationError(
      `Invalid negative or non-finite durationMs: ${durationMs}`
    );
  }
  if (durationMs === 0) {
    return 1;
  }
  return Math.max(1, Math.ceil((durationMs / 1000) * STUDIO_FPS - 1e-12));
};

/**
 * Snaps any positive integer frame count or 1-based start boundary onto the
 * causal 4n+1 lattice (1, 5, 9, 13, ...) using strict round-up arithmetic.
 */
export const conformFramesTo4nPlus1Lattice = (rawFrames: number): CausalLatticeResult => {
  if (!Number.isInteger(rawFrames) || rawFrames < 0) {
    throw new TemporalQuantizationError(
      `Invalid frame count for 4n+1 lattice conformance: ${rawFrames}`
    );
  }
  if (rawFrames <= 1) {
    return Object.freeze({
      rawFrames,
      latticeIndexN: 0,
      latticeFrames: 1,
      tailSlackFrames: 1 - rawFrames,
      conformsTo4nPlus1: true,
    });
  }

  const n = Math.ceil((rawFrames - 1) / 4);
  const latticeFrames = 4 * n + 1;
  const tailSlackFrames = latticeFrames - rawFrames;

  return Object.freeze({
    rawFrames,
    latticeIndexN: n,
    latticeFrames,
    tailSlackFrames,
    conformsTo4nPlus1: (latticeFrames - 1) % 4 === 0,
  });
};

/**
 * Snaps a 0-based start frame index onto the next 1-based 4n+1 causal lattice
 * boundary so (F_1based - 1) % 4 === 0 holds strictly.
 */
export const snapStartFrameTo4nPlus1 = (zeroBasedStartFrame: number): {
  zeroBasedAlignedStart: number;
  oneBasedLatticeFrame: number;
  preRollPadFrames: number;
} => {
  const oneBasedCandidate = zeroBasedStartFrame + 1;
  const lattice = conformFramesTo4nPlus1Lattice(oneBasedCandidate);
  const oneBasedLatticeFrame = lattice.latticeFrames;
  const zeroBasedAlignedStart = oneBasedLatticeFrame - 1;
  const preRollPadFrames = zeroBasedAlignedStart - zeroBasedStartFrame;

  return {
    zeroBasedAlignedStart,
    oneBasedLatticeFrame,
    preRollPadFrames,
  };
};

/**
 * Decouples a shot into pre_roll_frames, measured_duration_frames, and post_roll_frames.
 */
export const computeThreePartShotFrames = (
  entry: ExposureTimingEntry
): ThreePartShotFrames => {
  const measuredDurationFrames = quantizeDurationMsToFrames(entry.durationMs);
  const preRollFrames =
    entry.preRollMs && entry.preRollMs > 0
      ? quantizeDurationMsToFrames(entry.preRollMs)
      : 0;
  const postRollFrames =
    entry.postRollMs && entry.postRollMs > 0
      ? quantizeDurationMsToFrames(entry.postRollMs)
      : 0;

  return Object.freeze({
    preRollFrames,
    measuredDurationFrames,
    postRollFrames,
    totalSequenceFrames: preRollFrames + measuredDurationFrames + postRollFrames,
  });
};

/**
 * Quantizes an ordered list of exposure sheet entries using cumulative-millisecond
 * zero-drift tiling and causal 4n+1 lattice alignment.
 */
export const quantizeExposureTimeline = (
  entries: readonly ExposureTimingEntry[]
): QuantizedEpisodeTimeline => {
  let cumulativeMs = 0;
  const shots: QuantizedTimelineShot[] = [];

  for (const entry of entries) {
    if (!entry.shotId || entry.shotId.trim().length === 0) {
      throw new TemporalQuantizationError('Encountered exposure entry with empty shotId');
    }
    if (!Number.isFinite(entry.durationMs) || entry.durationMs < 0) {
      throw new TemporalQuantizationError(
        `Invalid durationMs (${entry.durationMs}) on shot '${entry.shotId}'`
      );
    }

    const baseThreePart = computeThreePartShotFrames(entry);
    const effectiveDurationMs =
      entry.durationMs < FRAME_DURATION_MS ? FRAME_DURATION_MS : entry.durationMs;
    const preMs = entry.preRollMs && entry.preRollMs > 0 ? entry.preRollMs : 0;
    const postMs = entry.postRollMs && entry.postRollMs > 0 ? entry.postRollMs : 0;

    const cumulativeStartMs = cumulativeMs;
    const startFrame =
      cumulativeStartMs === 0
        ? 0
        : Math.ceil((cumulativeStartMs / 1000) * STUDIO_FPS - 1e-12);

    // Optional 4n+1 lattice alignment on visual cut points
    let extraPreRollFrames = 0;
    let effectiveStartFrame = startFrame;
    if (entry.alignStartToLattice) {
      const aligned = snapStartFrameTo4nPlus1(startFrame);
      effectiveStartFrame = aligned.zeroBasedAlignedStart;
      extraPreRollFrames = aligned.preRollPadFrames;
    }

    const shotSpanMs = preMs + effectiveDurationMs + postMs + extraPreRollFrames * FRAME_DURATION_MS;
    const cumulativeEndMs = cumulativeStartMs + shotSpanMs;
    const computedEndFrame =
      Math.ceil((cumulativeEndMs / 1000) * STUDIO_FPS - 1e-12) - 1;
    const endFrame = Math.max(effectiveStartFrame, computedEndFrame);

    const totalSequenceFrames = endFrame - effectiveStartFrame + 1;
    const threePart: ThreePartShotFrames = Object.freeze({
      preRollFrames: baseThreePart.preRollFrames + extraPreRollFrames,
      measuredDurationFrames: baseThreePart.measuredDurationFrames,
      postRollFrames: Math.max(
        0,
        totalSequenceFrames -
          (baseThreePart.preRollFrames + extraPreRollFrames + baseThreePart.measuredDurationFrames)
      ),
      totalSequenceFrames,
    });

    const latticeConformance = conformFramesTo4nPlus1Lattice(totalSequenceFrames);
    const oneBasedLatticeStart = snapStartFrameTo4nPlus1(effectiveStartFrame).oneBasedLatticeFrame;

    shots.push(
      Object.freeze({
        shotId: entry.shotId,
        ...(entry.speakerId ? { speakerId: entry.speakerId } : {}),
        durationMs: entry.durationMs,
        cumulativeStartMs,
        cumulativeEndMs,
        startFrame: effectiveStartFrame,
        endFrame,
        latticeStartFrame1Based: oneBasedLatticeStart,
        threePart,
        latticeConformance,
      })
    );

    cumulativeMs = cumulativeEndMs;
  }

  const totalDeliveryFrames =
    shots.length === 0 ? 0 : shots[shots.length - 1].endFrame + 1;
  const renderedDurationMs = totalDeliveryFrames * FRAME_DURATION_MS;
  const maxDriftMs = Math.abs(renderedDurationMs - cumulativeMs);

  return Object.freeze({
    fps: STUDIO_FPS,
    shotCount: shots.length,
    totalDurationMs: cumulativeMs,
    totalDeliveryFrames,
    maxDriftMs,
    shots: Object.freeze(shots),
  });
};

export const initTemporal = (
  cpy: TemporalModel,
  bal: TemporalBit,
  _ste?: State
): TemporalModel => {
  cpy.fps = 24;
  cpy.lastTimeline = null;
  cpy.lastLatticeResult = null;
  cpy.quantizedShotCount = 0;
  cpy.temporalBits = {};
  if (bal.slv) bal.slv({ tmpBit: { idx: 'init-temporal', val: 1 } });
  return cpy;
};

export const updateTemporal = (
  cpy: TemporalModel,
  bal: TemporalBit,
  _ste?: State
): TemporalModel => {
  if (bal.idx) cpy.temporalBits[bal.idx] = bal;
  if (bal.slv) bal.slv({ tmpBit: bal });
  return cpy;
};

export const quantizeExposureSheet = (
  cpy: TemporalModel,
  bal: TemporalBit,
  _ste?: State
): TemporalModel => {
  const entries = bal.entries ?? [];
  const timeline = quantizeExposureTimeline(entries);
  cpy.lastTimeline = timeline;
  cpy.quantizedShotCount = timeline.shotCount;

  if (bal.slv) {
    bal.slv({
      tmpBit: {
        idx: bal.idx || 'quantize-exposure-sheet',
        val: timeline.totalDeliveryFrames,
        timeline,
      },
    });
  }
  return cpy;
};

export const conformLatticeBudget = (
  cpy: TemporalModel,
  bal: TemporalBit,
  _ste?: State
): TemporalModel => {
  const rawFrames = bal.val ?? 1;
  const lattice = conformFramesTo4nPlus1Lattice(rawFrames);
  cpy.lastLatticeResult = lattice;

  if (bal.slv) {
    bal.slv({
      tmpBit: {
        idx: bal.idx || 'conform-lattice-budget',
        val: lattice.latticeFrames,
        lattice,
      },
    });
  }
  return cpy;
};
