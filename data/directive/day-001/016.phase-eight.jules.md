```typescript
import * as fs from 'fs';
import * as path from 'path';
import * as crypto from 'crypto';
import { SealsBit } from '../fce/seals.bit';
import {
  LifecycleState,
  HeaderDocblockMetadata,
  ParsedSourceFile,
  SealEmissionInput,
  ClobberArbitrationResult
} from '../fce/seals.interface';

/**
 * Normalizes string text to standard POSIX Line Feeds (\n),
 * stripping carriage returns (\r\n -> \n) and trailing whitespace.
 */
export const normalizeLf = (text: string): string => {
  return text.replace(/\r\n/g, '\n').replace(/\r/g, '\n');
};

/**
 * [TASK-VT-8.1] Normalized Body Checksum Hash Calculator.
 * Calculates cryptographic SHA-256 digest over normalized body text.
 */
export const computeBodyChecksum = (bodyText: string): string => {
  const normalized = normalizeLf(bodyText).trim();
  return crypto.createHash('sha256').update(normalized, 'utf8').digest('hex');
};

/**
 * [TASK-VT-8.2] Lifecycle Header Docblock Formatter.
 * Synthesizes header comment docblock stamped at byte offset zero.
 */
export const formatHeaderDocblock = (meta: HeaderDocblockMetadata): string => {
  const timestamp = meta.sealedAt || new Date().toISOString();
  const patches = meta.patchHistory && meta.patchHistory.length > 0
    ? `\n// @patch_history: ${JSON.stringify(meta.patchHistory)}`
    : '';

  return `// @lifecycle: ${meta.lifecycle}
// @checksum: ${meta.checksum}
// @series: ${meta.seriesSlug}
// @episode: ${meta.episodeId}
// @sealed_at: ${timestamp}${patches}
`;
};

/**
 * Parses existing file to extract stamped metadata and isolated body text.
 */
export const parseExistingSourceFile = (content: string): ParsedSourceFile => {
  const normalized = normalizeLf(content);
  const headerRegex = /^\/\/\s*@lifecycle:\s*(generated|patched|locked)\n\/\/\s*@checksum:\s*([a-f0-9]{64})\n\/\/\s*@series:\s*([^\n]+)\n\/\/\s*@episode:\s*([^\n]+)(?:\n\/\/\s*@sealed_at:\s*([^\n]+))?(?:\n\/\/\s*@patch_history:\s*(\[[^\]]*\]))?\n+/;

  const match = normalized.match(headerRegex);
  if (!match) {
    return {
      hasHeader: false,
      bodyText: normalized,
      fullContent: normalized
    };
  }

  const [fullHeaderMatch, lifecycle, checksum, seriesSlug, episodeId, sealedAt, rawPatches] = match;
  const bodyText = normalized.slice(fullHeaderMatch.length);

  return {
    hasHeader: true,
    metadata: {
      lifecycle: lifecycle as LifecycleState,
      checksum,
      seriesSlug,
      episodeId,
      sealedAt,
      patchHistory: rawPatches ? JSON.parse(rawPatches) : undefined
    },
    bodyText,
    fullContent: normalized
  };
};

/**
 * [TASK-VT-8.3] Clobber Refusal Guard & Directorial Override Protector.
 * Arbitrates file emission: if target file is locked or manually modified,
 * writes to .candidate.tsx rather than clobbering existing source.
 */
export const arbitrateAndEmitComposition = (input: SealEmissionInput): ClobberArbitrationResult => {
  const normalizedBody = normalizeLf(input.rawBodySource).trim();
  const newChecksum = computeBodyChecksum(normalizedBody);

  const targetDir = path.dirname(input.targetFilePath);
  if (!fs.existsSync(targetDir)) {
    fs.mkdirSync(targetDir, { recursive: true });
  }

  const fileExists = fs.existsSync(input.targetFilePath);

  if (fileExists && !input.forceOverwrite) {
    const existingContent = fs.readFileSync(input.targetFilePath, 'utf8');
    const parsed = parseExistingSourceFile(existingContent);

    if (parsed.hasHeader && parsed.metadata) {
      const isLocked = parsed.metadata.lifecycle === 'locked';
      const actualExistingBodyChecksum = computeBodyChecksum(parsed.bodyText);
      const isDiverged = actualExistingBodyChecksum !== parsed.metadata.checksum;

      // Directorial Override Protection Tripwire
      if (isLocked || isDiverged) {
        const candidatePath = input.targetFilePath.replace(/\.tsx$/, '.candidate.tsx');
        const candidateHeader = formatHeaderDocblock({
          lifecycle: 'generated',
          checksum: newChecksum,
          seriesSlug: input.seriesSlug,
          episodeId: input.episodeId
        });

        fs.writeFileSync(candidatePath, `${candidateHeader}\n${normalizedBody}\n`, 'utf8');

        const reason = isLocked
          ? 'ERR_CLOBBER_REFUSED: File is marked @lifecycle: locked.'
          : 'ERR_CLOBBER_REFUSED: File checksum diverged from header. Manual edits detected.';

        return {
          status: 'REFUSED_EMITTED_CANDIDATE',
          targetPath: input.targetFilePath,
          writtenPath: candidatePath,
          lifecycle: parsed.metadata.lifecycle,
          checksum: newChecksum,
          divergedFromStamped: isDiverged,
          refusalReason: reason
        };
      }
    }
  }

  // Safe to write or overwrite cleanly
  const header = formatHeaderDocblock({
    lifecycle: 'generated',
    checksum: newChecksum,
    seriesSlug: input.seriesSlug,
    episodeId: input.episodeId
  });

  const fullOutput = `${header}\n${normalizedBody}\n`;
  fs.writeFileSync(input.targetFilePath, fullOutput, 'utf8');

  return {
    status: 'WRITTEN',
    targetPath: input.targetFilePath,
    writtenPath: input.targetFilePath,
    lifecycle: 'generated',
    checksum: newChecksum,
    divergedFromStamped: false
  };
};

export const runEmitSealedCompositionBuzz = (
  ste: any,
  bale: SealsBit,
  bal: { slv: Function; blk: Function }
): void => {
  try {
    const input = bale.dat?.input;
    if (!input) {
      throw new Error('ERR_MISSING_INPUT: bale.dat.input is required.');
    }

    const arbitration = arbitrateAndEmitComposition(input);

    if (arbitration.status === 'REFUSED_EMITTED_CANDIDATE') {
      bal.slv({
        idx: bale.idx,
        val: 1,
        src: `>> [REFUSAL] Directorial override detected. Emitted candidate draft: ${arbitration.writtenPath}`,
        dat: {
          receipt: {
            status: 'SUCCESS',
            episodeId: input.episodeId,
            arbitration
          }
        }
      });
      return;
    }

    bal.slv({
      idx: bale.idx,
      val: 1,
      src: `>> [OK] [SEAL] Stamped and sealed Episode ${input.episodeId} (checksum: ${arbitration.checksum.substring(0, 12)}...).`,
      dat: {
        receipt: {
          status: 'SUCCESS',
          episodeId: input.episodeId,
          arbitration
        }
      }
    });
  } catch (err: any) {
    bal.slv({
      idx: bale.idx,
      val: 0,
      src: `>> [FAIL] [SEAL] Invariant violation: ${err.message}`,
      dat: { error: err.message }
    });
  }
};
```

```typescript
import { describe, it, expect, beforeEach, afterEach } from 'vitest';
import * as fs from 'fs';
import * as path from 'path';
import {
  normalizeLf,
  computeBodyChecksum,
  formatHeaderDocblock,
  parseExistingSourceFile,
  arbitrateAndEmitComposition
} from '../buz/seals.buzz';

describe('19.seals.unit: Cryptographic Sealing & Collision Arbitration', () => {
  const tmpDir = path.resolve(__dirname, '.tmp_seals_test');
  const targetFile = path.join(tmpDir, 'Episode_040.tsx');
  const candidateFile = path.join(tmpDir, 'Episode_040.candidate.tsx');

  const baseSource = `export const Episode_040: React.FC = () => {
  return <div style={{ width: 1920, height: 1080 }} />;
};`;

  beforeEach(() => {
    fs.rmSync(tmpDir, { recursive: true, force: true });
    fs.mkdirSync(tmpDir, { recursive: true });
  });

  afterEach(() => {
    fs.rmSync(tmpDir, { recursive: true, force: true });
  });

  it('[TASK-VT-8.1] normalizes line-endings and generates bit-identical SHA-256 body checksums', () => {
    const windowsText = 'const x = 1;\r\nconst y = 2;\r\n';
    const posixText = 'const x = 1;\nconst y = 2;\n';

    const hash1 = computeBodyChecksum(windowsText);
    const hash2 = computeBodyChecksum(posixText);

    expect(hash1).toBe(hash2);
    expect(hash1).toMatch(/^[a-f0-9]{64}$/);

    // Single character mutation must alter the hash completely
    const mutated = 'const x = 1;\nconst y = 3;\n';
    const hashMutated = computeBodyChecksum(mutated);
    expect(hashMutated).not.toBe(hash1);
  });

  it('[TASK-VT-8.2] formats lifecycle header docblock and parses metadata accurately', () => {
    const checksum = computeBodyChecksum(baseSource);
    const header = formatHeaderDocblock({
      lifecycle: 'generated',
      checksum,
      seriesSlug: 'under-the-floorboards',
      episodeId: '040',
      sealedAt: '2026-11-04T14:30:00.000Z'
    });

    expect(header.startsWith('// @lifecycle: generated')).toBe(true);
    expect(header).toContain(`// @checksum: ${checksum}`);
    expect(header).toContain('// @series: under-the-floorboards');
    expect(header).toContain('// @episode: 040');

    const parsed = parseExistingSourceFile(`${header}\n${baseSource}`);
    expect(parsed.hasHeader).toBe(true);
    expect(parsed.metadata?.lifecycle).toBe('generated');
    expect(parsed.metadata?.checksum).toBe(checksum);
    expect(parsed.bodyText.trim()).toBe(baseSource.trim());
  });

  it('[TASK-VT-8.3] writes clean file when target does not yet exist', () => {
    const result = arbitrateAndEmitComposition({
      targetFilePath: targetFile,
      seriesSlug: 'under-the-floorboards',
      episodeId: '040',
      rawBodySource: baseSource
    });

    expect(result.status).toBe('WRITTEN');
    expect(result.writtenPath).toBe(targetFile);
    expect(fs.existsSync(targetFile)).toBe(true);

    const content = fs.readFileSync(targetFile, 'utf8');
    expect(content).toContain('// @lifecycle: generated');
    expect(content).toContain(baseSource);
  });

  it('[TASK-VT-8.3] clobber refusal guard refuses overwrite when file is @lifecycle: locked', () => {
    // 1. Initial write with locked state
    const checksum = computeBodyChecksum(baseSource);
    const lockedHeader = formatHeaderDocblock({
      lifecycle: 'locked',
      checksum,
      seriesSlug: 'under-the-floorboards',
      episodeId: '040'
    });
    fs.writeFileSync(targetFile, `${lockedHeader}\n${baseSource}\n`, 'utf8');

    // 2. Upstream compilation tries to regenerate
    const newBodySource = `${baseSource}\n// upstream automated tweak`;
    const result = arbitrateAndEmitComposition({
      targetFilePath: targetFile,
      seriesSlug: 'under-the-floorboards',
      episodeId: '040',
      rawBodySource: newBodySource
    });

    expect(result.status).toBe('REFUSED_EMITTED_CANDIDATE');
    expect(result.writtenPath).toBe(candidateFile);
    expect(fs.existsSync(candidateFile)).toBe(true);

    // Assert original target file was left completely untouched
    const untouched = fs.readFileSync(targetFile, 'utf8');
    expect(untouched).toContain('// @lifecycle: locked');
    expect(untouched).not.toContain('// upstream automated tweak');
  });

  it('[TASK-VT-8.3] clobber refusal guard triggers when body diverges from stamped checksum (manual edit detection)', () => {
    // 1. Initial write
    arbitrateAndEmitComposition({
      targetFilePath: targetFile,
      seriesSlug: 'under-the-floorboards',
      episodeId: '040',
      rawBodySource: baseSource
    });

    // 2. Director hits "Eject Button" and makes a manual manual edit without updating the header checksum
    const manuallyEdited = fs.readFileSync(targetFile, 'utf8').replace(
      'width: 1920',
      'width: 1920, transform: "scale(1.05)"'
    );
    fs.writeFileSync(targetFile, manuallyEdited, 'utf8');

    // 3. Automated re-compile occurs
    const recompileResult = arbitrateAndEmitComposition({
      targetFilePath: targetFile,
      seriesSlug: 'under-the-floorboards',
      episodeId: '040',
      rawBodySource: baseSource
    });

    expect(recompileResult.status).toBe('REFUSED_EMITTED_CANDIDATE');
    expect(recompileResult.divergedFromStamped).toBe(true);
    expect(fs.existsSync(candidateFile)).toBe(true);

    // Existing manual edits preserved
    const preserved = fs.readFileSync(targetFile, 'utf8');
    expect(preserved).toContain('transform: "scale(1.05)"');
  });

  it('guarantees pure 7-bit ASCII constraints across docblocks and outputs', () => {
    const nonAsciiRegex = /[^\x00-\x7F]/;
    const header = formatHeaderDocblock({
      lifecycle: 'generated',
      checksum: 'e3b0c44298fc1c149afbf4c8996fb92427ae41e4649b934ca495991b7852b855',
      seriesSlug: 'under-the-floorboards',
      episodeId: '040'
    });

    expect(nonAsciiRegex.test(header)).toBe(false);
  });
});
```
